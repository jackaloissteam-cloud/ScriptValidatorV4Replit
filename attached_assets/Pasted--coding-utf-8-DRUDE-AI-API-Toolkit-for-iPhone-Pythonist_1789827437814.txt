def load_key(pid):
    if keychain is None: return ''
    try: return (keychain.get_password(SERVICE,SERVICE+'.'+pid) or '').strip()
    except: return ''
def save_key(pid,val):
    if keychain is None: raise RuntimeError('Pythonista-Keychain nicht verfügbar.')
    keychain.set_password(SERVICE,SERVICE+'.'+pid,val.strip())
def del_key(pid):
    try: keychain.delete_password(SERVICE,SERVICE+'.'+pid)
    except: pass
def mask(v): return 'nicht eingerichtet' if not v else (v[:5]+'••••••••'+v[-4:] if len(v)>10 else '••••••••')
def detail(r):
    try:
        d=r.json()
        if isinstance(d,dict):
            for k in ('error','message','detail'):
                if k in d:
                    x=d[k]; return (x if isinstance(x,str) else json.dumps(x,ensure_ascii=False))[:220]
        return json.dumps(d,ensure_ascii=False)[:220]
    except: return (r.text or '')[:220]
def OK(msg='Key akzeptiert',extra=None): return {'ok':True,'detail':msg,'extra':extra}
def NO(msg): return {'ok':False,'detail':msg,'extra':None}

def test_hf(k):

def test_fal(k):
    # v2.2 FAL KEYCHECK:
    k=(k or '').strip()
    if ':' not in k:
        return NO(
            'Unvollständiger fal.ai-Key: vermutlich nur die Key-ID. '
            'Bitte einen neuen Key erstellen und den vollständigen Wert '
            'key_id:key_secret direkt beim Erstellen kopieren.'
        )
    left,right=k.split(':',1)
    if not left.strip() or len(right.strip()) < 8:
        return NO('fal.ai-Key sieht unvollständig aus. Erwartet: key_id:key_secret.')

    # v2.1 FAL PATCH:
    # 1) Ein normaler API-scope Key wird zuerst über den Pricing-Endpunkt geprüft.
    #    Dieser Aufruf erzeugt KEIN Bild und verursacht daher keine Inference-Kosten.
    # 2) Erst wenn der Key gültig ist, versuchen wir zusätzlich Billing/Credits.
    #    Billing benötigt bei fal.ai einen ADMIN-scope Key.
    auth={'Authorization':'Key '+k,'Accept':'application/json'}

    r=requests.get(
        'https://api.fal.ai/v1/models/pricing',
        headers=auth,
        params={'endpoint_id':'fal-ai/flux/dev'},
        timeout=TIMEOUT
    )

    if r.status_code!=200:
        return NO(f'HTTP {r.status_code}: {detail(r)}')

    pricing=r.json()

    # Key ist ab hier sicher akzeptiert. Billing ist nur Bonusinformation.
    try:
        b=requests.get(
            'https://api.fal.ai/v1/account/billing',
            headers=auth,
            params={'expand':'credits'},
            timeout=TIMEOUT
        )

        if b.status_code==200:
            d=b.json()
            c=d.get('credits')
            username=d.get('username')

            msg='Key gültig'
            if username:
                msg += ' · '+str(username)

            if isinstance(c,dict):
                balance=c.get('current_balance', c.get('balance', c.get('amount')))
                currency=c.get('currency','')
                if balance is not None:
                    msg += f' · Guthaben {balance} {currency}'.rstrip()

            return OK(msg, {'pricing':pricing,'billing':d})

        if b.status_code in (401,403):
            return OK('Key gültig · API-Scope · Billing benötigt ADMIN-Key', pricing)

        return OK(f'Key gültig · Billing HTTP {b.status_code}', pricing)

    except Exception:
        return OK('Key gültig · API-Scope', pricing)

            ui.delay(lambda:console.alert('Restart ausgelöst','Der Space wird neu gestartet.','OK',hide_cancel_button=True),.05)
        except Exception as e:ui.delay(lambda:console.alert('Restart Fehler',str(e),'OK',hide_cancel_button=True),.05)
    threading.Thread(target=run,daemon=True).start()
